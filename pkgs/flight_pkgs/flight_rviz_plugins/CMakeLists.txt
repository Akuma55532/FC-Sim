cmake_minimum_required(VERSION 3.8)
project(flight_rviz_plugins)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(flight_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
find_package(yaml-cpp REQUIRED)
add_library(waypoint_file STATIC src/waypoint_file.cpp)
set_target_properties(waypoint_file PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(waypoint_file PUBLIC include)
target_link_libraries(waypoint_file yaml-cpp)
add_library(flight_panel SHARED src/flight_panel.cpp include/flight_rviz_plugins/flight_panel.hpp)
target_include_directories(flight_panel PUBLIC include)
target_link_libraries(flight_panel Qt5::Widgets waypoint_file)
ament_target_dependencies(flight_panel rviz_common pluginlib rclcpp rclcpp_action flight_interfaces geometry_msgs visualization_msgs std_srvs)
pluginlib_export_plugin_description_file(rviz_common plugin_description.xml)
install(TARGETS flight_panel waypoint_file ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_waypoint_file test/test_waypoint_file.cpp)
  target_link_libraries(test_waypoint_file waypoint_file)
  ament_add_gtest(test_trajectory test/test_trajectory.cpp)
  target_include_directories(test_trajectory PRIVATE include)
  ament_target_dependencies(test_trajectory geometry_msgs)
  ament_add_gtest(test_quadrotor_model test/test_quadrotor_model.cpp)
  target_include_directories(test_quadrotor_model PRIVATE include)
  ament_target_dependencies(test_quadrotor_model visualization_msgs)
  ament_add_gtest(test_panel test/test_panel.cpp)
  target_link_libraries(test_panel flight_panel Qt5::Widgets)
  ament_target_dependencies(test_panel rviz_common pluginlib rclcpp rclcpp_action flight_interfaces visualization_msgs)
endif()
ament_package()
